function gerarPdfDaLicao(iframe) { var win = iframe.contentWindow; var doc = iframe.contentDocument || (win && win.document); var src = doc && doc.getElementById('licaoImpressao'); if (!src) { alert('Não foi possível gerar o PDF. Tente novamente.'); return; } if (typeof html2pdf !== 'function') { alert('Não foi possível carregar o gerador de PDF. Verifique a internet.'); return; } try { if (typeof win.gravarCache === 'function') win.gravarCache(); } catch (e) {} var antigo = document.getElementById('__pdfCloneWrap'); if (antigo) antigo.remove(); var wrap = document.createElement('div'); wrap.id = '__pdfCloneWrap'; wrap.style.cssText = 'position:fixed;left:0;top:0;width:794px;background:#fff;z-index:2147483646;pointer-events:none;'; var clone = src.cloneNode(true); clone.id = 'licaoImpressaoClone'; clone.style.cssText = 'width:794px;max-width:794px;margin:0;box-shadow:none;border-radius:0;background:#fff;'; clone.querySelectorAll('#btnPdf, #btnCorrigir, .btn-pdf, .btn-corrigir').forEach(function (el) { el.remove(); }); clone.querySelectorAll('img').forEach(function (img) { img.removeAttribute('crossorigin'); img.src = 'logo-ipb.png'; img.style.position = 'relative'; img.style.left = 'auto'; img.style.height = '52px'; }); var origCampos = src.querySelectorAll('input, textarea, select'); var cloneCampos = clone.querySelectorAll('input, textarea, select'); origCampos.forEach(function (inp, i) { var c = cloneCampos[i]; if (!c) return; if (inp.type === 'checkbox' || inp.type === 'radio') c.checked = inp.checked; else c.value = inp.value; if (inp.parentElement && c.parentElement) { c.parentElement.className = inp.parentElement.className; } }); wrap.appendChild(clone); document.body.appendChild(wrap); var nome = 'Aluno'; try { if (typeof win.val === 'function') nome = String(win.val('nomePessoa') || 'Aluno'); } catch (e) {} nome = String(nome).replace(/\s+/g, '_'); var nLicao = '01'; try { var m = String((win.location && win.location.pathname) || '').match(/Licao(\d+)/i); if (m) nLicao = m[1]; } catch (e) {} html2pdf().set({ margin: [8, 10, 8, 10], filename: 'Licao' + nLicao + '-' + nome + '.pdf', image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: false, allowTaint: false, backgroundColor: '#ffffff', logging: false, scrollX: 0, scrollY: 0, windowWidth: 794, imageTimeout: 0 }, pagebreak: { mode: ['avoid-all', 'css', 'legacy'], avoid: ['.question-group', '.section-title', '.final-note', '.alternativas', '.intro', '.header-container'] }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }).from(clone).save().then(function () { wrap.remove(); }).catch(function () { wrap.remove(); alert('Não foi possível gerar o PDF. Tente novamente.'); }); }
